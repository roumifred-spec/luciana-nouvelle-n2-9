def _nom_de_tiroir_refuse(nom):
    """Rend la raison du refus, ou "" si le nom est bon."""
    t = _sans_accents(str(nom or "")).strip()
    if not t:
        return "un nom de tiroir est vide"
    if t in _NOMS_DE_TIROIR_INTERDITS:
        return "'%s' n'est pas un sujet, c'est un fourre-tout" % nom
    if _NOM_DE_TIROIR_DATE.search(t):
        return ("'%s' COMMENCE par une date (du genre 06/10, '6 octobre', 'octobre 2026') : un "
                "tiroir se nomme par son sujet. Mets le sujet devant ; une date plus loin dans le "
                "nom ne pose aucun probleme" % nom)
    return ""
